import { adapterFor } from "../adapters";
import { getSettings } from "./settings";
import type { BulkConversationItem, BulkExportResult, Conversation, ExportFormat, ProviderId } from "./types";
import { buildConversationFromBulk } from "../content/bulk-builder";
import { logger } from "./logger";

export interface BulkExportOptions {
  providerId: ProviderId;
  conversationIds: string[];
  formats: ExportFormat[];

export async function fetchConversationList(
  providerId: ProviderId,
  limit: number = 30,
  offset: number = 0
): Promise<{ items: BulkConversationItem[]; nextOffset?: number; total?: number }> {
  const adapter = adapterFor(new URL(`https://${providerId}.com`));
  if (!adapter) {
    throw new Error(`No adapter for provider: ${providerId}`);
  }

  if (!adapter.fetchList) {
    throw new Error(`Bulk fetch not supported for: ${providerId}`);
  }

  // Pobierz orgId/authToken jeśli potrzebne
  let authContext: string | null = null;
  if (adapter.getOrgId) {
    authContext = adapter.getOrgId(document);
  } else if (adapter.getAuthToken) {
    authContext = adapter.getAuthToken(document);
  }

  if (!authContext && adapter.supportsBulk) {
    logger.warn(`No auth context found for ${providerId}, trying without...`);
  }

  const result = await adapter.fetchList(authContext || "", limit, offset);
  
  // Adapter powinien zwrócić { items: BulkConversationItem[], nextOffset?: number }
  return result;
}

export async function fetchConversationDetail(
  providerId: ProviderId,
  conversationId: string
): Promise<any> {
  const adapter = adapterFor(new URL(`https://${providerId}.com`));
  if (!adapter) {
    throw new Error(`No adapter for provider: ${providerId}`);
  }

  if (!adapter.fetchDetail) {
    throw new Error(`Bulk detail fetch not supported for: ${providerId}`);
  }

  let authContext: string | null = null;
  if (adapter.getOrgId) {
    authContext = adapter.getOrgId(document);
  } else if (adapter.getAuthToken) {
    authContext = adapter.getAuthToken(document);
  }

  return await adapter.fetchDetail(authContext || "", conversationId);
}

export async function runBulkExport(
  options: BulkExportOptions
): Promise<BulkExportResult[]> {
  const { providerId, conversationIds, formats, saveAttachments = false } = options;
  const results: BulkExportResult[] = [];
  const settings = await getSettings();

  // Używamy przekazanych formatów lub domyślnych z ustawień
  const exportFormats = formats.length > 0 ? formats : settings.enabledFormats;

  for (const id of conversationIds) {
    try {
      logger.info(`Bulk exporting conversation: ${id}`);
      
      // Pobierz szczegóły
      const detail = await fetchConversationDetail(providerId, id);
      
      // Zbuduj Conversation z surowych danych
      const conv = await buildConversationFromBulk(providerId, detail, {
        saveAttachments,
        attachmentsFolder: settings.attachmentsFolder,
      });

      // Wyślij do background do zapisu
      const response = await chrome.runtime.sendMessage({
        kind: "save-conversation",
        conversation: conv,
        formats: exportFormats,
      });

      if (response?.ok) {
        results.push({
          id,
          title: conv.title,
          success: true,
          formats: exportFormats,
          filename: conv.title,
        });
      } else {
        results.push({
          id,
          title: conv.title || id,
          success: false,
          error: response?.error || "Unknown save error",
        });
      }
    } catch (err: any) {
      logger.error(`Bulk export failed for ${id}`, err);
      results.push({
        id,
        title: id,
        success: false,
        error: err.message || "Unknown error",
      });
    }
  }

  return results;
}

import type { Conversation, Part, ProviderId, Attachment } from "../core/types";
import { adapterFor } from "../adapters";
import { logger } from "../core/logger";

export interface BulkBuildOptions {
  saveAttachments?: boolean;
  attachmentsFolder?: string;
}

export async function buildConversationFromBulk(
  providerId: ProviderId,
  data: any,
  options: BulkBuildOptions = {}
): Promise<Conversation> {
  const adapter = adapterFor(new URL(`https://${providerId}.com`));
  if (!adapter) {
    throw new Error(`No adapter for provider: ${providerId}`);
  }

  // Adapter musi mieć metodę parseBulkData
  if (!adapter.parseBulkData) {
    throw new Error(`Adapter ${providerId} does not support bulk parsing`);
  }

  const parsed = await adapter.parseBulkData(data, options);

  const conv: Conversation = {
    schemaVersion: 1,
    provider: providerId,
    title: parsed.title || "Untitled conversation",
    url: parsed.url || "",
    capturedAt: new Date().toISOString(),
    messages: parsed.messages || [],
    warnings: parsed.warnings || [],
    attachments: parsed.attachments || [],
  };

  if (parsed.chatId) conv.chatId = parsed.chatId;
  if (parsed.providerModel) conv.providerModel = parsed.providerModel;

  return conv;
}

function getOrgIdFromPage(doc: Document): string | null {
  // Metoda 1: z HTML
  const html = doc.documentElement.innerHTML;
  const match = html.match(/https:\/\/claude\.ai\/api\/organizations\/([a-f0-9-]{36})/);
  if (match?.[1]) return match[1];

  // Metoda 2: z sessionStorage
  try {
    const statuses = JSON.parse(sessionStorage.getItem('SSS-cardamom-integration-statuses') || '{}');
    if (statuses?.orgUuid) return statuses.orgUuid;
  } catch {}

  return null;
}

export const claudeAdapter: ProviderAdapter = {
  // ... istniejące metody

  // BULK
  getOrgId(doc) {
    return getOrgIdFromPage(doc);
  },

  supportsBulk: true,

  async fetchList(authContext: string, limit: number, offset: number) {
    const orgId = authContext;
    if (!orgId) throw new Error("Organization ID required for Claude bulk");

    const url = `https://claude.ai/api/organizations/${orgId}/chat_conversations?limit=${limit}&offset=${offset}`;
    const response = await fetch(url, {
      credentials: "include",
      headers: { accept: "*/*" },
    });

    if (!response.ok) {
      throw new Error(`Claude API error: ${response.status} ${response.statusText}`);
    }

    const data = await response.json();
    return {
      items: data.map((item: any) => ({
        id: item.uuid,
        title: item.name || "Untitled",
        url: `https://claude.ai/chat/${item.uuid}`,
        createdAt: item.created_at,
        updatedAt: item.updated_at,
      })),
      nextOffset: data.length === limit ? offset + limit : undefined,
      total: data.length,
    };
  },

  async fetchDetail(authContext: string, conversationId: string) {
    const orgId = authContext;
    if (!orgId) throw new Error("Organization ID required for Claude bulk");

    const url = `https://claude.ai/api/organizations/${orgId}/chat_conversations/${conversationId}?tree=True&rendering_mode=messages&render_all_tools=true`;
    const response = await fetch(url, {
      credentials: "include",
      headers: { accept: "*/*" },
    });

    if (!response.ok) {
      throw new Error(`Claude API error: ${response.status} ${response.statusText}`);
    }

    return await response.json();
  },

  async parseBulkData(data: any, options: any = {}) {
    const messages: any[] = [];
    const attachments: Attachment[] = [];
    const warnings: string[] = [];
    const artifactMemory: Record<string, any> = {};

    const chatMessages = data.chat_messages || [];

    for (const msg of chatMessages) {
      const role = msg.sender === "human" ? "user" 
                  : msg.sender === "assistant" ? "assistant" 
                  : "tool";
      const contentParts: string[] = [];

      for (const content of msg.content || []) {
        if (!content) continue;

        // Tekst
        if (content.type === "text" && content.text) {
          contentParts.push(content.text);
          continue;
        }

        // Tool result
        if (content.type === "tool_result") {
          const result = typeof content.content === "string" 
            ? content.content 
            : content.content?.[0]?.text || content.text || "";
          if (result) contentParts.push(result);
          continue;
        }

        // Tool use – artefakty!
        if (content.type === "tool_use") {
          const toolName = content.name || "";
          const input = content.input || {};

          if (toolName === "artifacts") {
            const artifactId = input.id || content.id || input.version_uuid || `artifact-${attachments.length + 1}`;
            const command = input.command || "create";
            const type = input.type || input.textdoc_type || "text/plain";
            const title = input.title || artifactId;

            let contentStr = "";

            if (command === "create" || command === "rewrite") {
              contentStr = typeof input.content === "string" 
                ? input.content 
                : JSON.stringify(input.content || input, null, 2);
            } else if (command === "update") {
              const current = artifactMemory[artifactId]?.current || "";
              const oldStr = input.old_str || "";
              const newStr = input.new_str || (typeof input.content === "string" ? input.content : "");
              if (current.includes(oldStr)) {
                contentStr = current.replace(oldStr, newStr);
              } else {
                contentStr = `${current}\n\n/* patch_failed: "${oldStr}" not found */\n${newStr}`;
              }
            } else {
              contentStr = typeof input.content === "string" 
                ? input.content 
                : JSON.stringify(input.content || input, null, 2);
            }

            const version = (artifactMemory[artifactId]?.version || 0) + 1;
            const finalId = input.version_uuid || `${artifactId}-v${version}`;

            // Zapisz jako załącznik jeśli opcja włączona
            if (options.saveAttachments) {
              const ext = getExtensionFromMime(type);
              attachments.push({
                name: `${title} (v${version}).${ext}`,
                url: `data:text/plain;base64,${btoa(unescape(encodeURIComponent(contentStr)))}`,
                mime: type,
                data: contentStr,
              });
            }

            artifactMemory[artifactId] = {
              current: contentStr,
              title: title,
              type: type,
              version: version,
            };

            contentParts.push(`**[Artifact: ${title} v${version}]**\n\n\`\`\`${getLanguageFromMime(type)}\n${contentStr}\n\`\`\``);
          } else {
            // Inne narzędzia
            contentParts.push(`**Tool: ${toolName}**\n\`\`\`json\n${JSON.stringify(input, null, 2)}\n\`\`\``);
          }
          continue;
        }

        if (content.text) contentParts.push(content.text);
      }

      const contentText = contentParts.filter(Boolean).join("\n\n") || " ";
      messages.push({
        id: msg.uuid,
        role: role,
        parts: [{ type: "text", markdown: contentText }],
        createdAt: msg.created_at,
      });
    }

    return {
      title: data.name || "Claude conversation",
      url: `https://claude.ai/chat/${data.uuid}`,
      chatId: data.uuid,
      providerModel: data.model || undefined,
      messages,
      attachments,
      warnings,
    };
  },
};

function getExtensionFromMime(mime: string): string {
  const map: Record<string, string> = {
    "text/plain": "txt",
    "text/markdown": "md",
    "text/html": "html",
    "application/json": "json",
    "text/x-python": "py",
    "text/javascript": "js",
    "text/typescript": "ts",
    "text/x-rust": "rs",
    "text/x-c": "c",
    "text/x-cpp": "cpp",
    "text/x-java": "java",
    "text/x-go": "go",
  };
  return map[mime] || mime.split("/").pop() || "txt";
}

function getLanguageFromMime(mime: string): string {
  const map: Record<string, string> = {
    "text/plain": "text",
    "text/markdown": "markdown",
    "text/html": "html",
    "application/json": "json",
    "text/x-python": "python",
    "text/javascript": "javascript",
    "text/typescript": "typescript",
    "text/x-rust": "rust",
    "text/x-c": "c",
    "text/x-cpp": "cpp",
    "text/x-java": "java",
    "text/x-go": "go",
  };
  return map[mime] || mime.split("/").pop() || "text";
}

function getGeminiAuthToken(doc: Document): { sid: string; at: string } | null {
  // Pobieramy z chrome.storage.local (zapisane przez content script)
  // To będzie działać tylko w content script
  try {
    // Używamy sync, bo to jest content script
    const result = chrome.storage.local.getSync?.('gemini_credentials_map');
    if (result?.gemini_credentials_map) {
      const map = result.gemini_credentials_map;
      const entries = Object.values(map);
      if (entries.length > 0) {
        // Bierzemy najnowszy
        const latest = entries.sort((a: any, b: any) => (b.lastUsed || 0) - (a.lastUsed || 0))[0];
        return { sid: latest.sid, at: latest.at || "" };
      }
    }
    return null;
  } catch {
    return null;
  }
}

export const geminiAdapter: ProviderAdapter = {
  // ... istniejące metody

  getAuthToken(doc) {

        createdAt: new Date(item.timestamp || Date.now()).toISOString(),
      })),
      nextOffset: data.nextPageToken ? offset + limit : undefined,
      total: data.conversations.length,
    };
  },

  async fetchDetail(authContext: string, conversationId: string) {
    const sid = authContext;
    if (!sid) throw new Error("SID required for Gemini bulk");

    const response = await chrome.runtime.sendMessage({
      action: "gemini-fetch-detail",
      payload: { sid, conversationId },
    });

    if (!response?.success) {
      throw new Error(response?.error || "Gemini fetch detail failed");

  async parseBulkData(data: any, options: any = {}) {
    const messages: any[] = [];
    const attachments: Attachment[] = [];

    const rawMessages = data.messages || [];

    for (const msg of rawMessages) {
      const role = msg.role === "user" ? "user" : "assistant";
      let content = msg.content || "";

      // Obrazy
      if (msg.images && msg.images.length > 0) {
        for (const img of msg.images) {
          if (options.saveAttachments) {
            attachments.push({
              name: img.fileName || `image-${Date.now()}.jpg`,
              url: img.sourceUrl || img.resolvedUrl || img.dataUrl || "",
              mime: img.mimeType || "image/jpeg",
              data: img.dataUrl || "",
            });
          }
          content += `\n\n![${img.fileName || "image"}](${img.resolvedUrl || img.sourceUrl || img.dataUrl || ""})`;
        }
      }

      // Dokumenty
      if (msg.documents && msg.documents.length > 0) {
        for (const doc of msg.documents) {
          const docContent = doc.contentMarkdown || doc.sections?.map((s: any) => s.content).join("\n\n") || "";
          if (options.saveAttachments && docContent) {
            attachments.push({
              name: doc.title || `document-${Date.now()}.md`,
              url: `data:text/plain;base64,${btoa(unescape(encodeURIComponent(docContent)))}`,
              mime: "text/markdown",
              data: docContent,
            });
          }
          content += `\n\n📎 **Document:** ${doc.title}\n\n${docContent}`;
        }
      }

      messages.push({
        role,
        parts: [{ type: "text", markdown: content }],
        createdAt: msg.timestamp ? new Date(msg.timestamp).toISOString() : undefined,
      });
    }

    return {
      title: data.title || "Gemini conversation",
      url: data.url || `https://gemini.google.com/app/${data.id?.replace(/^c_/, "") || ""}`,
      chatId: data.id?.replace(/^c_/, ""),
      messages,
      attachments,
      warnings: [],
    };
  },
};

// --- ATTACHMENTS SUPPORT ---

function extractDeepSeekAttachments(doc: Document): Attachment[] {
  const attachments: Attachment[] = [];
  
  // Szukamy załączników w wiadomościach
  const imageSelectors = [
    'img[src*="attachment"]',
    'img[src*="upload"]',
    'img[data-type="attachment"]',
    '.ds-message img:not([src*="emoji"])',
  ];

  for (const selector of imageSelectors) {
    const imgs = doc.querySelectorAll(selector);
    for (const img of imgs) {
      const src = (img as HTMLImageElement).src;
      if (src && !src.startsWith("data:") && !src.includes("emoji")) {
        const name = img.getAttribute("alt") || `attachment-${attachments.length + 1}.png`;
        attachments.push({
          name,
          url: src,
          mime: "image/*",
        });
      }
    }
  }

  // Pliki (linki)
  const fileSelectors = [
    'a[href*="file"]',
    'a[download]',
    '[data-testid*="attachment"] a',
  ];

  for (const selector of fileSelectors) {
    const links = doc.querySelectorAll(selector);
    for (const link of links) {
      const href = (link as HTMLAnchorElement).href;
      const name = link.textContent?.trim() || link.getAttribute("download") || `file-${attachments.length + 1}`;
      if (href && !attachments.find(a => a.url === href)) {
        attachments.push({
          name,
          url: href,
          mime: "application/octet-stream",
        });
      }
    }
  }

  return attachments;
}

// Rozszerzamy adapter – nadpisujemy extract
const originalExtract = deepseekAdapter.extract;
deepseekAdapter.extract = function(doc) {
  const messages = originalExtract(doc);
  const attachments = extractDeepSeekAttachments(doc);
  
  // Dodajemy załączniki do pierwszej wiadomości (lub jako osobna)
  if (attachments.length > 0 && messages.length > 0) {
    const lastMsg = messages[messages.length - 1];
    const attachmentText = attachments.map(a => 
      `📎 **Załącznik:** [${a.name}](${a.url})`
    ).join("\n");
    lastMsg.parts.push({ type: "text", markdown: attachmentText });
  }
  
  return messages;
};

import { getSettings } from "../core/settings";
import { logger } from "../core/logger";

// Funkcja do pobierania załącznika
async function fetchAttachment(url: string): Promise<Blob> {
  // Jeśli to data URL
  if (url.startsWith("data:")) {
    const response = await fetch(url);
    return await response.blob();
  }

  // Jeśli to URL zewnętrzny
  const response = await fetch(url, {
    credentials: "include",
    headers: { accept: "*/*" },
  });

  if (!response.ok) {
    throw new Error(`Failed to fetch attachment: ${response.status}`);
  }

  return await response.blob();
}

// Funkcja do zapisywania załącznika
async function saveAttachment(
  blob: Blob,
  filename: string,
  folder: string
): Promise<string> {
  const safeFilename = filename.replace(/[^a-zA-Z0-9._-]/g, "_");
  const fullPath = `${folder}/${safeFilename}`;

  // Sprawdzamy, czy plik już istnieje – jeśli tak, dodajemy numer
  const url = URL.createObjectURL(blob);
  try {
    await chrome.downloads.download({
      url,
      filename: fullPath,
      conflictAction: "uniquify",
      saveAs: false,
    });
  } catch (err: any) {
    logger.error("Failed to save attachment", err);
    throw err;
  } finally {
    URL.revokeObjectURL(url);
  }

  return fullPath;
}

// Zmiana w obsłudze save-conversation
chrome.runtime.onMessage.addListener((msg: RuntimeMessage, _sender, sendResponse) => {
  if (msg.kind === "save-conversation") {
    (async () => {
      try {
        const settings = await getSettings();
        const conv = msg.conversation;
        let firstFilename = "";

        // Zapisz załączniki (jeśli włączone)
        if (settings.saveAttachments && conv.attachments?.length) {
          const folder = settings.attachmentsFolder || "attachments";
          for (const att of conv.attachments) {
            try {
              let blob: Blob;
              if (att.data) {
                // Jeśli mamy już dane (z bulk)
                if (typeof att.data === "string") {
                  blob = new Blob([att.data], { type: att.mime || "text/plain" });
                } else {
                  blob = att.data as Blob;
                }
              } else if (att.url) {
                blob = await fetchAttachment(att.url);
              } else {
                continue;
              }

              const savedPath = await saveAttachment(blob, att.name, folder);
              att.savedPath = savedPath;
              
              // Dodaj informację o załączniku do ostatniej wiadomości
              if (conv.messages.length > 0) {
                const lastMsg = conv.messages[conv.messages.length - 1];
                const attachmentNote = `\n\n📎 **Załącznik:** [${att.name}](${savedPath})`;
                if (lastMsg.parts.length > 0) {
                  const lastPart = lastMsg.parts[lastMsg.parts.length - 1];
                  if (lastPart.type === "text") {
                    lastPart.markdown += attachmentNote;
                  }
                }
              }
            } catch (err: any) {
              logger.error(`Failed to save attachment ${att.name}`, err);
              conv.warnings.push(`Failed to save attachment: ${att.name} - ${err.message}`);
            }
          }
        }

        // Zapisz rozmowę w wybranych formatach
        for (const fmt of msg.formats) {
          const fn = await downloadOne(conv, fmt, settings);
          if (!firstFilename) firstFilename = fn;
        }

        // ... reszta (historia itp.)
        await pushHistory({
          id: Date.now().toString(),
          at: conv.capturedAt,
          provider: conv.provider,
          title: conv.title,
          url: conv.url,
          chatId: conv.chatId,
          filename: firstFilename,
          formats: msg.formats,
          warnings: conv.warnings.length,
          messageCount: conv.messages.length,
          attachmentCount: conv.attachments?.length || 0,
        });

        sendResponse({ ok: true });
      } catch (err: any) {
        logger.error("Save failed", err.message);
        sendResponse({ ok: false, error: err.message });
      }
    })();
    return true;
  }
});

async function onProviderChange(e: Event) {
  const select = e.target as HTMLSelectElement;
  const settings = await getSettings();
  settings.bulkProvider = select.value as ProviderId;
  await saveSettings(settings);
  await onFetch();

async function onFetch() {
  if (isRunning) return;

  const providerSelect = document.getElementById("provider") as HTMLSelectElement;
  const provider = providerSelect.value as ProviderId;
  const limit = parseInt((document.getElementById("limit") as HTMLInputElement)?.value || "30");

  const statusEl = document.getElementById("status");
  const listEl = document.getElementById("conversation-list");

  if (!listEl) return;

  try {
    isRunning = true;
    if (statusEl) statusEl.textContent = "Loading...";

    const result = await fetchConversationList(provider, limit);
    conversations = result.items;
    selectedIds = new Set();

    renderList(listEl, conversations);

    if (statusEl) {
      statusEl.textContent = `Loaded ${conversations.length} conversations`;
    }
  } catch (err: any) {
    logger.error("Failed to fetch conversations", err);
    if (statusEl) statusEl.textContent = `Error: ${err.message}`;
  } finally {
    isRunning = false;
  }
}

function renderList(container: HTMLElement, items: BulkConversationItem[]) {
  container.innerHTML = "";

  if (items.length === 0) {
    container.innerHTML = '<div class="empty">No conversations found</div>';
    return;
  }

  for (const item of items) {
    const div = document.createElement("div");
    div.className = "conversation-item";
    div.innerHTML = `
      <input type="checkbox" data-id="${item.id}" ${selectedIds.has(item.id) ? "checked" : ""}>
      <span class="title">${item.title || "Untitled"}</span>
      <span class="date">${new Date(item.createdAt).toLocaleDateString()}</span>
    `;
    const checkbox = div.querySelector("input") as HTMLInputElement;
    checkbox.addEventListener("change", () => {
      if (checkbox.checked) {
        selectedIds.add(item.id);
      } else {
        selectedIds.delete(item.id);
      }
      updateExportButton();
    });
    container.appendChild(div);
  }
}

function toggleAll(selected: boolean) {
  selectedIds = new Set();
  if (selected) {
    for (const item of conversations) {
      selectedIds.add(item.id);
    }
  }

  const container = document.getElementById("conversation-list");
  if (container) {
    const checkboxes = container.querySelectorAll('input[type="checkbox"]');
    for (const cb of checkboxes) {
      (cb as HTMLInputElement).checked = selected;
    }
  }

  updateExportButton();
}

function updateExportButton() {
  const btn = document.getElementById("export-btn") as HTMLButtonElement;
  if (btn) {
    btn.textContent = `Export ${selectedIds.size} conversation${selectedIds.size !== 1 ? "s" : ""}`;
    btn.disabled = selectedIds.size === 0 || isRunning;
  }
}

async function onExport() {
  if (isRunning || selectedIds.size === 0) return;

  const providerSelect = document.getElementById("provider") as HTMLSelectElement;
  const provider = providerSelect.value as ProviderId;

  // Format
  const formatCheckboxes = document.querySelectorAll('input[name="format"]:checked');
  const formats = Array.from(formatCheckboxes).map(el => (el as HTMLInputElement).value as ExportFormat);

  // Opcje
  const saveAttachments = (document.getElementById("save-attachments") as HTMLInputElement)?.checked || false;

  const statusEl = document.getElementById("status");
  const progressEl = document.getElementById("progress");

  try {
    isRunning = true;
    if (statusEl) statusEl.textContent = "Exporting...";
    if (progressEl) progressEl.textContent = "0%";

    const result = await runBulkExport({
      providerId: provider,
      conversationIds: Array.from(selectedIds),
      formats: formats.length > 0 ? formats : ["md"],
      saveAttachments,
    });

    const successCount = result.filter(r => r.success).length;
    const failCount = result.filter(r => !r.success).length;

    if (statusEl) {
      statusEl.textContent = `Done: ${successCount} saved, ${failCount} failed`;
    }

    if (failCount > 0) {
      const errors = result.filter(r => !r.success).map(r => `${r.title}: ${r.error}`);
      console.error("Bulk export errors:", errors);
    }

    // Odśwież listę
    await onFetch();

  } catch (err: any) {
    logger.error("Bulk export failed", err);
    if (statusEl) statusEl.textContent = `Error: ${err.message}`;
  } finally {
    isRunning = false;
    updateExportButton();
    if (progressEl) progressEl.textContent = "";
  }
}

// src/adapters/kimi.ts – DODAJ na końcu pliku

// --- ATTACHMENTS SUPPORT FOR KIMI ---

function extractKimiAttachments(doc: Document): Attachment[] {
  const attachments: Attachment[] = [];
  
  // 1. Obrazy w wiadomościach
  const imgSelectors = [
    '.chat-content-item-user img:not([src*="emoji"])',
    '.chat-content-item-assistant img:not([src*="emoji"])',
    'img[src*="upload"]',
    'img[data-type="attachment"]',
  ];
  
  for (const selector of imgSelectors) {
    const imgs = doc.querySelectorAll(selector);
    for (const img of imgs) {
      const src = (img as HTMLImageElement).src;
      if (src && !src.startsWith("data:") && !src.includes("emoji")) {
        attachments.push({
          name: img.getAttribute("alt") || `image-${attachments.length + 1}.png`,
          url: src,
          mime: "image/*",
        });
      }
    }
  }
  
  // 2. Pliki (linki do pobrania)
  const fileSelectors = [
    'a[href*="download"]',
    'a[href*="file"]',
    'a[download]',
    '[class*="file"] a',
    '[data-testid*="attachment"] a',
  ];
  
  for (const selector of fileSelectors) {
    const links = doc.querySelectorAll(selector);
    for (const link of links) {
      const href = (link as HTMLAnchorElement).href;
      const name = link.textContent?.trim() || link.getAttribute("download") || `file-${attachments.length + 1}`;
      if (href && !attachments.find(a => a.url === href)) {
        attachments.push({
          name,
          url: href,
          mime: "application/octet-stream",
        });
      }
    }
  }
  
  // 3. Iframe'y z treścią (już masz extractIframeContent)
  // Możemy to wykorzystać do wyciągnięcia zawartości iframe jako załącznika
  const iframes = doc.querySelectorAll('iframe');
  for (const iframe of iframes) {
    try {
      const iframeDoc = iframe.contentDocument || iframe.contentWindow?.document;
      if (iframeDoc) {
        const text = iframeDoc.body?.textContent?.trim();
        if (text && text.length > 100) {
          const src = iframe.src || iframe.getAttribute("src") || "unknown";
          attachments.push({
            name: `iframe-content-${attachments.length + 1}.txt`,
            url: src,
            mime: "text/plain",
            data: text,
          });
        }
      }
    } catch {
      // Cross-origin iframe – pomijamy
    }
  }
  
  return attachments;
}

// Rozszerzamy adapter – nadpisujemy extract
const originalKimiExtract = kimiAdapter.extract;
kimiAdapter.extract = function(doc) {
  const messages = originalKimiExtract(doc);
  const attachments = extractKimiAttachments(doc);
  
  // Dodajemy załączniki do ostatniej wiadomości (jeśli są)
  if (attachments.length > 0 && messages.length > 0) {
    const lastMsg = messages[messages.length - 1];
    const attachmentText = attachments.map(a => 
      `📎 **Załącznik:** [${a.name}](${a.url})`
    ).join("\n");
    lastMsg.parts.push({ type: "text", markdown: attachmentText });
  }
  
  return messages;
};

// --- BULK SUPPORT FOR KIMI ---

kimiAdapter.supportsBulk = true;

kimiAdapter.getOrgId = function(doc: Document): string | null {
  // Kimi nie ma orgId, ale możemy użyć sessionId lub userId
  try {
    // Próbujemy wyciągnąć userId z localStorage lub sessionStorage
    const userData = localStorage.getItem('user_info') || sessionStorage.getItem('user_info');
    if (userData) {
      const parsed = JSON.parse(userData);
      return parsed.userId || parsed.id || null;
    }
  } catch {}
  
  // Alternatywnie – z HTML
  const html = doc.documentElement.innerHTML;
  const match = html.match(/userId["']?\s*[:=]\s*["']([^"']+)["']/i);
  if (match?.[1]) return match[1];
  
  return null;
};

kimiAdapter.fetchList = async function(authContext: string, limit: number, offset: number): Promise<any> {
  // Kimi API – trzeba ustalić endpoint
  // Na podstawie analizy strony Kimi, wygląda na to, że używają GraphQL lub REST
  // Przykład (do zweryfikowania):
  const url = `https://kimi.com/api/conversations?limit=${limit}&offset=${offset}`;
  
  const response = await fetch(url, {
    credentials: "include",
    headers: {
      "accept": "application/json",
      "x-user-id": authContext || "",
    },
  });
  
  if (!response.ok) {
    // Jeśli API nie działa, możemy spróbować wyciągnąć listę z DOM
    return fetchKimiListFromDOM(limit);
  }
  
  const data = await response.json();
  return {
    items: data.conversations?.map((item: any) => ({
      id: item.id,
      title: item.title || "Untitled",
      url: `https://kimi.com/chat/${item.id}`,
      createdAt: item.createdAt || new Date().toISOString(),
    })) || [],
    nextOffset: data.hasMore ? offset + limit : undefined,
    total: data.total || data.conversations?.length || 0,
  };
};

// Fallback – lista z DOM (jeśli API nie działa)
async function fetchKimiListFromDOM(limit: number): Promise<any> {
  const items: any[] = [];
  
  // Szukamy listy rozmów w DOM
  const selectors = [
    '.chat-list-item',
    '.conversation-item',
    '[class*="chat-item"]',
    '[data-conversation-id]',
  ];
  
  for (const selector of selectors) {
    const elements = document.querySelectorAll(selector);
    for (const el of elements) {
      const id = el.getAttribute('data-conversation-id') || 
                 el.getAttribute('data-id') || 
                 el.id;
      const title = el.querySelector('.title, .name, [class*="title"]')?.textContent?.trim() || 
                    el.textContent?.trim() || 
                    "Untitled";
      
      if (id) {
        items.push({
          id,
          title: title.slice(0, 100),
          url: `https://kimi.com/chat/${id}`,
          createdAt: new Date().toISOString(),
        });
      }
    }
    
    if (items.length >= limit) break;
  }
  
  return {

kimiAdapter.fetchDetail = async function(authContext: string, conversationId: string): Promise<any> {
  // Próba przez API
  try {
    const url = `https://kimi.com/api/conversations/${conversationId}`;
    const response = await fetch(url, {
      credentials: "include",
      headers: {
        "accept": "application/json",
        "x-user-id": authContext || "",
      },
    });
    
    if (response.ok) {
      return await response.json();
    }
  } catch {}
  
  // Fallback – wyciągamy z DOM (jeśli jesteśmy na stronie rozmowy)
  // Ale to nie zadziała w bulk, bo bulk działa na osobnej stronie
  throw new Error("Cannot fetch Kimi conversation detail – please implement DOM parsing");
};

kimiAdapter.parseBulkData = async function(data: any, options: any = {}): Promise<any> {
  const messages: any[] = [];
  const attachments: Attachment[] = [];
  
  // Jeśli mamy dane z API
  if (data.messages || data.chat_messages) {
    const rawMessages = data.messages || data.chat_messages || [];
    for (const msg of rawMessages) {
      const role = msg.role === "user" ? "user" : "assistant";
      const content = msg.content || msg.text || "";
      
      // Szukamy załączników w treści
      if (options.saveAttachments) {
        const extracted = extractAttachmentsFromContent(content);
        attachments.push(...extracted);
      }
      
      messages.push({
        role,
        parts: [{ type: "text", markdown: content }],
        createdAt: msg.createdAt || msg.timestamp,
      });
    }
  } else {
    // Fallback – próbujemy wyciągnąć z DOM
    // (to nie zadziała w bulk, ale zostawiamy jako placeholder)
    throw new Error("Cannot parse Kimi data – unknown format");
  }
  
  return {
    title: data.title || "Kimi conversation",
    url: data.url || `https://kimi.com/chat/${data.id}`,
    chatId: data.id,
    messages,
    attachments,
    warnings: [],
  };
};

// Helper – wyciąga załączniki z treści (np. markdown z obrazami)
function extractAttachmentsFromContent(content: string): Attachment[] {
  const attachments: Attachment[] = [];
  
  // Szukamy obrazów w markdown
  const imgRegex = /!\[([^\]]*)\]\(([^)]+)\)/g;
  let match;
  while ((match = imgRegex.exec(content)) !== null) {
    const name = match[1] || `image-${attachments.length + 1}.png`;
    const url = match[2];
    attachments.push({
      name,
      url,
      mime: "image/*",
    });
  }
  
  // Szukamy linków do plików
  const linkRegex = /\[([^\]]+)\]\(([^)]+)\)/g;
  while ((match = linkRegex.exec(content)) !== null) {
    const name = match[1];
    const url = match[2];
    if (url.match(/\.(pdf|doc|docx|xls|xlsx|ppt|pptx|txt|zip|rar|7z|png|jpg|jpeg|gif|webp|svg|mp4|mp3|wav|avi|mov)/i)) {
      attachments.push({
        name,
        url,
        mime: "application/octet-stream",
      });
    }
  }
  
  return attachments;
}
2. Czy to zadziała od razu?
Załączniki w Kimi – tak, jeśli:

Załączniki są w formie obrazów <img> lub linków <a> w DOM.

Są w iframe'ach (już masz extractIframeContent).

Bulk w Kimi – to jest trudniejsze, ponieważ:

Kimi może nie mieć publicznego API (lub wymaga tokena).

Lista rozmów może być dostępna tylko po zalogowaniu na stronie.

Rozwiązanie dla Bulk w Kimi:

Opcja A – API – jeśli Kimi ma API, użyjemy go (trzeba znaleźć endpoint).

Opcja B – DOM – jeśli nie ma API, możemy wyciągnąć listę rozmów z DOM (ale to działa tylko na stronie kimi.com z listą rozmów).

Opcja C – Hybryda – próbujemy API, jeśli nie działa – używamy DOM.

W praktyce, dla Kimi proponuję:

async function extractIframeContent(container: Element): Promise<Part[]> {
  const parts: Part[] = [];
  const iframes = container.querySelectorAll("iframe");
  
  for (const iframe of iframes) {
    try {
      const iframeDoc = iframe.contentDocument || iframe.contentWindow?.document;
      if (iframeDoc) {
        const text = iframeDoc.body?.textContent?.trim();
        if (text && text.length > 50) {
          const src = iframe.src || iframe.getAttribute("src") || "unknown";
          parts.push({ 
            type: "text", 
            markdown: `**Iframe content (${src}):**\n\n${text}` 
          });
        }
      }
    } catch {
      // Cross-origin iframe - can't access content
      const src = iframe.src || iframe.getAttribute("src") || "unknown";
      parts.push({ 
        type: "text", 
        markdown: `**Iframe detected (cross-origin, cannot extract):** ${src}` 
      });
    }
  }
  return parts;
}

async function extractSidePanelContent(doc: Document): Promise<Part[]> {
  const parts: Part[] = [];
  
  // Look for side panel / file preview containers
  const sidePanelSelectors = [
    '[class*="side-panel"]',
    '[class*="file-preview"]',
    '[class*="artifact-panel"]',
    '[class*="sandbox"]',
    '[class*="preview-panel"]',
    '.chat-sidebar',
    '[data-testid*="side"]',
    '[data-testid*="preview"]',
  ];
  
  for (const selector of sidePanelSelectors) {
    const panels = doc.querySelectorAll(selector);
    for (const panel of panels) {
      const text = panel.textContent?.trim();
      if (text && text.length > 50) {
        // Check if this looks like file content (has code, markdown, etc.)
        if (/```|\.md|function|class|import|def |const |# /.test(text)) {
          parts.push({ 
            type: "text", 
            markdown: `**Side panel content:**\n\n${text}` 
          });
        }
      }
    }
  }
  return parts;
}

function extractKimiAttachments(doc: Document): Attachment[] {
  const attachments: Attachment[] = [];
  const seen = new Set<string>();
  
  // 1. Obrazy w wiadomościach (użytkownik + asystent)
  const imgSelectors = [
    '.chat-content-item-user img:not([src*="emoji"]):not([src*="avatar"])',
    '.chat-content-item-assistant img:not([src*="emoji"]):not([src*="avatar"])',
    '.markdown-container img:not([src*="emoji"])',
    'img[src*="upload"]',
    'img[src*="attachment"]',
    'img[data-type="attachment"]',
    'img[class*="file"]',
    'img[class*="image"]',
  ];
  
  for (const selector of imgSelectors) {
    const imgs = doc.querySelectorAll(selector);
    for (const img of imgs) {
      const src = (img as HTMLImageElement).src;
      if (src && !src.startsWith("data:") && !src.includes("emoji") && !src.includes("avatar") && !seen.has(src)) {
        seen.add(src);
        const name = img.getAttribute("alt") || img.getAttribute("title") || `image-${attachments.length + 1}.png`;
        attachments.push({
          name,
          url: src,
          mime: "image/*",
          size: (img as HTMLImageElement).naturalWidth || undefined,
        });
      }
    }
  }
  
  // 2. Pliki (linki do pobrania)
  const fileSelectors = [
    'a[href*="download"]',
    'a[href*="file"]',
    'a[download]',
    '[class*="file"] a',
    '[data-testid*="attachment"] a',
    '.attachment-link',
    '.file-link',
    'a[href*=".pdf"]',
    'a[href*=".doc"]',
    'a[href*=".zip"]',
    'a[href*=".txt"]',
  ];
  
  for (const selector of fileSelectors) {
    const links = doc.querySelectorAll(selector);
    for (const link of links) {
      const href = (link as HTMLAnchorElement).href;
      if (href && !seen.has(href)) {
        seen.add(href);
        const name = link.textContent?.trim() || 
                     link.getAttribute("download") || 
                     link.getAttribute("title") || 
                     `file-${attachments.length + 1}`;
        
        // Określ MIME na podstawie rozszerzenia
        const mime = getMimeFromUrl(href);
        attachments.push({
          name: name.replace(/[^a-zA-Z0-9._-]/g, "_"),
          url: href,
          mime,
        });
      }
    }
  }
  
  // 3. Iframe'y z treścią (już masz extractIframeContent, ale dodajemy jako załączniki)
  const iframes = doc.querySelectorAll('iframe');
  for (const iframe of iframes) {
    try {
      const iframeDoc = iframe.contentDocument || iframe.contentWindow?.document;
      if (iframeDoc) {
        const text = iframeDoc.body?.textContent?.trim();
        if (text && text.length > 100) {
          const src = iframe.src || iframe.getAttribute("src") || "unknown";
          const name = `iframe-${attachments.length + 1}.txt`;
          attachments.push({
            name,
            url: src,
            mime: "text/plain",
            data: text,
          });
        }
      }
    } catch {
      // Cross-origin – pomijamy
    }
  }
  
  // 4. Canvas / SVG (rzadkie, ale mogą być)
  const canvasSelectors = ['canvas', 'svg'];
  for (const selector of canvasSelectors) {
    const elements = doc.querySelectorAll(selector);
    for (const el of elements) {
      if (el instanceof HTMLCanvasElement) {
        try {
          const dataUrl = el.toDataURL('image/png');
          attachments.push({
            name: `canvas-${attachments.length + 1}.png`,
            url: dataUrl,
            mime: "image/png",
            data: dataUrl,
          });
        } catch {}
      }
    }
  }
  
  return attachments;
}

// Pobierz userId / auth context
function getKimiAuthContext(doc: Document): string | null {
  // Próba 1: localStorage
  try {
    const userData = localStorage.getItem('user_info') || localStorage.getItem('user');
    if (userData) {
      const parsed = JSON.parse(userData);
      return parsed.userId || parsed.id || parsed.user_id || null;
    }
  } catch {}
  
  // Próba 2: sessionStorage
  try {
    const userData = sessionStorage.getItem('user_info') || sessionStorage.getItem('user');
    if (userData) {
      const parsed = JSON.parse(userData);
      return parsed.userId || parsed.id || parsed.user_id || null;
    }
  } catch {}
  
  // Próba 3: z HTML (meta lub script)
  const html = doc.documentElement.innerHTML;
  const patterns = [
    /userId["']?\s*[:=]\s*["']([^"']+)["']/i,
    /user_id["']?\s*[:=]\s*["']([^"']+)["']/i,
    /"id"\s*:\s*"([^"]+)"[^}]*"avatar"/i,
  ];
  for (const pattern of patterns) {
    const match = html.match(pattern);
    if (match?.[1]) return match[1];
  }
  
  return null;
}

// Pobierz listę rozmów z DOM (sidebar)
function fetchKimiListFromDOM(limit: number): any[] {
  const items: any[] = [];
  const seen = new Set<string>();
  
  // 1. Szukamy w sidebarze
  const sidebarSelectors = [
    '.chat-sidebar',

    '.sidebar',
    '[class*="sidebar"]',
    '[data-testid="conversation-list"]',
    '.chat-history',
    '.history-list',
  ];
  
  let sidebar: Element | null = null;
  for (const selector of sidebarSelectors) {
    sidebar = document.querySelector(selector);
    if (sidebar) break;
  }
  
  if (sidebar) {
    // Szukamy elementów rozmów
    const itemSelectors = [
      '[data-conversation-id]',
      '[data-id]',
      '.chat-item',
      '.conversation-item',
      '.history-item',
      '[class*="conversation"]',
      '[class*="chat-item"]',
      'li[class*="item"]',
    ];
    
    for (const selector of itemSelectors) {
      const elements = sidebar.querySelectorAll(selector);
      for (const el of elements) {
        const id = el.getAttribute('data-conversation-id') || 
                   el.getAttribute('data-id') || 
                   el.id;
        
        if (id && !seen.has(id)) {
          seen.add(id);
          const title = el.querySelector('.title, .name, [class*="title"], [class*="name"]')?.textContent?.trim() || 
                        el.textContent?.trim() || 
                        "Untitled";
          
          const url = `https://kimi.com/chat/${id}`;
          const createdAt = el.getAttribute('data-created-at') || 
                           el.getAttribute('data-time') || 
                           new Date().toISOString();
          
          items.push({ id, title: title.slice(0, 200), url, createdAt });
          
          if (items.length >= limit) break;
        }
      }
      if (items.length >= limit) break;
    }
  }
  
  // 2. Jeśli nie znaleziono w sidebarze – szukamy w całym dokumencie
  if (items.length === 0) {
    const allSelectors = [
      '[data-conversation-id]',
      '[data-id]',
      '.conversation-item',
      '.chat-item',
    ];
    
    for (const selector of allSelectors) {
      const elements = document.querySelectorAll(selector);
      for (const el of elements) {
        const id = el.getAttribute('data-conversation-id') || 
                   el.getAttribute('data-id') || 
                   el.id;
        
        if (id && !seen.has(id) && !id.includes('sidebar') && !id.includes('menu')) {
          seen.add(id);
          const title = el.textContent?.trim() || "Untitled";
          items.push({
            id,
            title: title.slice(0, 200),
            url: `https://kimi.com/chat/${id}`,
            createdAt: new Date().toISOString(),
          });
          
          if (items.length >= limit) break;
        }
      }
      if (items.length >= limit) break;
    }
  }
  
  return items;
}

export const kimiAdapter: ProviderAdapter = {
  id: "kimi",
  match: (u) => /(^|\.)kimi\.com$/.test(u.hostname),
  isFullyExpandedView: (u) => /\/share\//.test(u.pathname),
  
  getTitle(doc) {
    const fromTitle = doc.title.replace(/\s*[-–|]\s*Kimi.*$/i, "").trim();
    if (fromTitle) return fromTitle;
    
    const fromName = doc.querySelector(".chat-name")?.textContent?.trim();
    return fromName || "Kimi conversation";
  },
  
  async expandAll(doc) {
    await expandUntilStable(doc, [
      'button[aria-expanded="false"]',
      'div[class*="thinking"] button',
      'div[class*="reason"] button',
      'button[class*="expand"]',
      '[class*="collapse"] button',
    ]);
  },
  
  // ======== GŁÓWNA EKSTRAKCJA ========
  async extract(doc) {
    logger.debug("Starting Kimi extraction", { url: doc.location?.href || "unknown" });
    
    const messages: Message[] = [];
    const allAttachments: Attachment[] = [];
    
    // 1. Znajdź wszystkie tury rozmowy
    const turns = Array.from(doc.querySelectorAll(".chat-content-item-user, .chat-content-item-assistant, .message-item, [class*="message"]"));
    logger.debug(`Found ${turns.length} turns`);
    
    for (const el of turns) {
      const isUser = el.classList.contains("chat-content-item-user") || 
                     el.classList.contains("user-message") ||
                     el.getAttribute("data-role") === "user";
      
      const role = isUser ? "user" : "assistant";
      
      // Znajdź treść
      let body = el.querySelector(".markdown-container, .message-content, .content, [class*="markdown"], [class*="content"]") ?? el;
      
      // Wyciągnij części
      let parts = filterUiChromeParts(nodeToParts(body));
      
      // Thinking / reasoning
      const thinking = el.querySelector('[class*="thinking"], [class*="reason"], [class*="reasoning"]');
      if (thinking && thinking !== body) {
        const t = thinking.textContent?.trim();
        if (t) {
          parts.unshift({ type: "thinking", markdown: t });
        }
      }
      
      // Iframe'y w tej wiadomości
      const iframeParts = await extractIframeContent(el);
      parts.push(...iframeParts);
      
      // Załączniki w tej wiadomości (obrazy, pliki)
      const msgAttachments = extractAttachmentsFromElement(el);
      allAttachments.push(...msgAttachments);
      
      // Dodaj informację o załącznikach do treści
      if (msgAttachments.length > 0) {
        const attachmentText = msgAttachments.map(a => 
          `📎 **Załącznik:** [${a.name}](${a.url})`
        ).join("\n");
        parts.push({ type: "text", markdown: attachmentText });
      }
      
      if (parts.length) {
        messages.push({ 
          role, 
          parts, 
          createdAt: readTimestamp(el) 
        });
      }
    }
    
    // 2. Ekstrakcja z paneli bocznych (artefakty, pliki)
    const sidePanelParts = await extractSidePanelContent(doc);
    if (sidePanelParts.length) {
      // Dodaj jako ostatnią wiadomość asystenta (lub osobną)
      const lastAssistant = messages.findLastIndex(m => m.role === "assistant");
      if (lastAssistant >= 0) {
        messages[lastAssistant].parts.push(...sidePanelParts);
      } else {
        messages.push({ 
          role: "assistant", 
          parts: sidePanelParts, 
          createdAt: new Date().toISOString() 
        });
      }
    }
    
    // 3. Dodaj załączniki z całego dokumentu (jeśli nie zostały dodane)
    const docAttachments = extractKimiAttachments(doc);
    const allAttachmentUrls = new Set(allAttachments.map(a => a.url));
    const missingAttachments = docAttachments.filter(a => !allAttachmentUrls.has(a.url));
    
    if (missingAttachments.length > 0) {
      // Dodaj do ostatniej wiadomości
      if (messages.length > 0) {
        const lastMsg = messages[messages.length - 1];
        const attachmentText = missingAttachments.map(a => 
          `📎 **Załącznik:** [${a.name}](${a.url})`
        ).join("\n");
        lastMsg.parts.push({ type: "text", markdown: attachmentText });
      }
    }
    
    logger.debug(`Kimi extraction complete: ${messages.length} messages, ${docAttachments.length} attachments`);
    return messages;
  },
  
  // ======== BULK SUPPORT ========
  supportsBulk: true,
  
  getOrgId(doc) {
    return getKimiAuthContext(doc);
  },
  
  getAuthToken(doc) {
    // Kimi używa userId zamiast tokena
    return getKimiAuthContext(doc);
  },
  
  async fetchList(authContext: string, limit: number, offset: number) {
    logger.debug("Kimi fetchList", { authContext, limit, offset });
    
    // 1. Próba przez API (jeśli dostępne)
    try {
      const url = `https://kimi.com/api/conversations?limit=${limit}&offset=${offset}`;
      const response = await fetch(url, {
        credentials: "include",
        headers: {
          "accept": "application/json",
          "x-user-id": authContext || "",
        },
      });
      
      if (response.ok) {
        const data = await response.json();
        logger.debug("Kimi API fetchList success", { count: data.conversations?.length });
        return {
          items: data.conversations?.map((item: any) => ({
            id: item.id,
            title: item.title || "Untitled",
            url: `https://kimi.com/chat/${item.id}`,
            createdAt: item.createdAt || item.created_at || new Date().toISOString(),
            updatedAt: item.updatedAt || item.updated_at,
          })) || [],
          nextOffset: data.hasMore || data.next_offset ? offset + limit : undefined,
          total: data.total || data.conversations?.length || 0,
        };
      }
    } catch (err) {
      logger.warn("Kimi API fetchList failed, falling back to DOM", err);
    }
    
    // 2. Fallback: DOM (działa tylko na stronie z listą rozmów)
    const items = fetchKimiListFromDOM(limit);
    const paginated = items.slice(offset, offset + limit);
    
    return {
      items: paginated,
      nextOffset: items.length > offset + limit ? offset + limit : undefined,
      total: items.length,
    };
  },
  
  async fetchDetail(authContext: string, conversationId: string) {
    logger.debug("Kimi fetchDetail", { authContext, conversationId });
    
    // 1. Próba przez API
    try {
      const url = `https://kimi.com/api/conversations/${conversationId}`;
      const response = await fetch(url, {
        credentials: "include",
        headers: {
          "accept": "application/json",
          "x-user-id": authContext || "",
        },
      });
      
      if (response.ok) {
        const data = await response.json();
        logger.debug("Kimi API fetchDetail success");
        return data;
      }
    } catch (err) {
      logger.warn("Kimi API fetchDetail failed, falling back to DOM", err);
    }
    
    // 2. Fallback: jeśli jesteśmy na stronie rozmowy, użyj extract()
    if (window.location.href.includes(`/chat/${conversationId}`)) {
      const messages = await this.extract(document);
      return {
        id: conversationId,
        title: this.getTitle(document),
        messages: messages,
        url: window.location.href,
        createdAt: new Date().toISOString(),
      };
    }
    
    throw new Error(`Cannot fetch Kimi conversation detail for ${conversationId} – please open the conversation first`);
  },
  
  async parseBulkData(data: any, options: any = {}) {
    logger.debug("Kimi parseBulkData", { hasData: !!data, options });
    
    const messages: any[] = [];
    const attachments: Attachment[] = [];
    const warnings: string[] = [];
    
    // Jeśli mamy dane z API
    if (data.messages || data.chat_messages) {
      const rawMessages = data.messages || data.chat_messages || [];
      
      for (const msg of rawMessages) {
        const role = msg.role === "user" || msg.sender === "user" ? "user" : "assistant";
        let content = msg.content || msg.text || msg.message || "";
        
        // Szukamy załączników w treści
        if (options.saveAttachments) {
          const extracted = extractAttachmentsFromContent(content);
          attachments.push(...extracted);
        }
        
        // Dodajemy informację o załącznikach
        const msgAttachments = extractAttachmentsFromContent(content);
        if (msgAttachments.length > 0 && options.saveAttachments) {
          content += "\n\n" + msgAttachments.map(a => 
            `📎 **Załącznik:** [${a.name}](${a.url})`
          ).join("\n");
        }
        
        messages.push({
          role,
          parts: [{ type: "text", markdown: content }],
          createdAt: msg.createdAt || msg.created_at || msg.timestamp,
        });
      }
    } 
    // Jeśli mamy dane z DOM (z extract)
    else if (data.messages && Array.isArray(data.messages)) {
      for (const msg of data.messages) {
        messages.push({
          role: msg.role || "assistant",
          parts: msg.parts || [{ type: "text", markdown: msg.content || "" }],
          createdAt: msg.createdAt,
        });
      }
    }
    // Jeśli mamy surowe dane (np. z iframe)
    else if (typeof data === 'string') {
      messages.push({
        role: "assistant",
        parts: [{ type: "text", markdown: data }],
        createdAt: new Date().toISOString(),
      });
    }
    
    // Wyciągnij załączniki z całego dokumentu (jeśli są dostępne)
    if (options.saveAttachments && typeof document !== 'undefined') {
      const docAttachments = extractKimiAttachments(document);
      attachments.push(...docAttachments);
    }
    
    return {
      title: data.title || this.getTitle(document) || "Kimi conversation",
      url: data.url || window.location.href || `https://kimi.com/chat/${data.id}`,
      chatId: data.id || data.conversationId,
      providerModel: data.model || data.providerModel,
      messages,
      attachments: attachments.filter((a, i, self) => 
        self.findIndex(b => b.url === a.url) === i
      ),
      warnings,
    };
  },
};

function extractAttachmentsFromElement(el: Element): Attachment[] {
  const attachments: Attachment[] = [];
  const seen = new Set<string>();
  
  // Obrazy
  const imgs = el.querySelectorAll('img:not([src*="emoji"]):not([src*="avatar"])');
  for (const img of imgs) {
    const src = (img as HTMLImageElement).src;
    if (src && !src.startsWith("data:") && !seen.has(src)) {
      seen.add(src);
      attachments.push({
        name: img.getAttribute("alt") || `image-${attachments.length + 1}.png`,
        url: src,
        mime: "image/*",
      });
    }
  }
  
  // Linki do plików
  const links = el.querySelectorAll('a[href*="download"], a[href*="file"], a[download]');
  for (const link of links) {
    const href = (link as HTMLAnchorElement).href;
    if (href && !seen.has(href)) {
      seen.add(href);
      attachments.push({
        name: link.textContent?.trim() || link.getAttribute("download") || `file-${attachments.length + 1}`,
        url: href,
        mime: getMimeFromUrl(href),
      });
    }
  }
  
  return attachments;
}

function extractAttachmentsFromContent(content: string): Attachment[] {
  const attachments: Attachment[] = [];
  const seen = new Set<string>();
  
  // Obrazy w markdown
  const imgRegex = /!\[([^\]]*)\]\(([^)]+)\)/g;
  let match;
  while ((match = imgRegex.exec(content)) !== null) {
    const name = match[1] || `image-${attachments.length + 1}.png`;
    const url = match[2];
    if (!seen.has(url)) {
      seen.add(url);
      attachments.push({
        name,
        url,
        mime: "image/*",
      });
    }
  }
  
  // Linki do plików
  const linkRegex = /\[([^\]]+)\]\(([^)]+)\)/g;
  while ((match = linkRegex.exec(content)) !== null) {
    const name = match[1];
    const url = match[2];
    if (url.match(/\.(pdf|doc|docx|xls|xlsx|ppt|pptx|txt|md|json|xml|html|css|js|ts|py|java|c|cpp|h|hpp|go|rs|swift|rb|php|sql|sh|bash|zip|rar|7z|tar|gz|png|jpg|jpeg|gif|webp|svg|bmp|ico|mp4|webm|mp3|wav|ogg|flac)/i) && !seen.has(url)) {
      seen.add(url);
      attachments.push({
        name,
        url,
        mime: getMimeFromUrl(url),
      });
    }
  }
  
  return attachments;
}

Co zmieniłem / dodałem?
Element	Opis
extractKimiAttachments()	Nowa funkcja – wykrywa obrazy, pliki, iframe'y, canvas/SVG w całym dokumencie
getMimeFromUrl()	Określa typ MIME na podstawie rozszerzenia pliku
getKimiAuthContext()	Pobiera userId z localStorage/sessionStorage/HTML
fetchKimiListFromDOM()	Pobiera listę rozmów z sidebaru (fallback dla API)
extractAttachmentsFromElement()	Wykrywa załączniki w pojedynczym elemencie (wiadomości)
extractAttachmentsFromContent()	Wykrywa załączniki w treści tekstowej (markdown)
Bulk methods	fetchList, fetchDetail, parseBulkData
Rozszerzona ekstrakcja	Dodaje informację o załącznikach do treści wiadomości
Jak to działa?
Załączniki:

Skanuje wszystkie obrazy, linki, iframe'y w wiadomościach.

Dodaje je jako osobne Attachment obiekty.

Wstawia informację o załączniku w treści (np. 📎 **Załącznik:** [image.png](url)).

Bulk export:

Próbuje użyć API Kimi (jeśli istnieje).

Jeśli API nie działa – używa DOM (sidebar z listą rozmów).

Dla szczegółów rozmowy – próbuje API, jeśli nie działa – używa extract() (ale to wymaga, żeby być na stronie rozmowy).

Fallback:

Wszystko ma fallback do DOM, więc nawet jeśli API nie działa, część funkcji będzie dostępna.

Co jeszcze trzeba zrobić?
Sprawdzić, czy Kimi ma API – jeśli tak, można zastąpić fallback DOM pełnym API.

Dostosować selektory – jeśli struktura DOM Kimi się zmieni, trzeba zaktualizować selektory w fetchKimiListFromDOM() i extractKimiAttachments().

Przetestować – najlepiej na żywej stronie Kimi.

Przykład użycia
typescript
// W content script – pobranie listy rozmów
const result = await kimiAdapter.fetchList(userId, 30, 0);
console.log(result.items); // lista rozmów

// Pobranie szczegółów
const detail = await kimiAdapter.fetchDetail(userId, conversationId);

// Parsowanie
const parsed = await kimiAdapter.parseBulkData(detail, { saveAttachments: true });
console.log(parsed.messages); // wiadomości z załącznikami
