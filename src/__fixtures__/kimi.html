Here's a complete Docker Compose setup for your stack:

version: '3.8'

services:
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    ports:
      - "3000:3000"
    environment:
      - REACT_APP_API_URL=http://localhost:5000
    depends_on:
      - backend
    volumes:
      - ./frontend:/app
      - /app/node_modules

  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    ports:
      - "5000:5000"
    environment:
      - MONGODB_URI=mongodb://db:27017/myapp
      - NODE_ENV=development
    depends_on:
      - db
    volumes:
      - ./backend:/app
      - /app/node_modules

  db:
    image: mongo:7
    ports:
      - "27017:27017"
    volumes:
      - mongo_data:/data/db

      MONGO_INITDB_ROOT_USERNAME: admin

How do I add a Redis cache layer?

11:05 AM

Add a Redis service to your docker-compose.yml:

  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"
    volumes:
      - redis_data:/data
    command: redis-server --appendonly yes

volumes:
  redis_data:

And update your backend to use Redis:

const redis = require('redis');
const client = redis.createClient({
  url: process.env.REDIS_URL || 'redis://redis:6379'
});

async function getCachedData(key) {
  const cached = await client.get(key);
  if (cached) return JSON.parse(cached);

  const data = await fetchFromDatabase(key);
  await client.setEx(key, 3600, JSON.stringify(data));
  return data;
}

I'll also mention RedisInsight for local debugging and suggest using ioredis over redis npm package for better performance.
