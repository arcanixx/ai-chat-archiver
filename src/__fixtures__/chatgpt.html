Python Data Pipeline Design

I need to design a data pipeline that processes CSV files, transforms them, and loads into a PostgreSQL database. What's the best approach?

10:00 AM

I recommend using a modular pipeline approach with these components:

  1. Extract — Use Pandas or Polars for CSV loading
  2. Transform — Apply cleaning and business logic
  3. Load — Use SQLAlchemy for database insertion

Here's a complete implementation:

import pandas as pd
from sqlalchemy import create_engine
from pathlib import Path
from typing import List, Dict, Any
import logging

logger = logging.getLogger(__name__)

class DataPipeline:
    def __init__(self, db_url: str, batch_size: int = 1000):
        self.engine = create_engine(db_url)
        self.batch_size = batch_size

    def extract(self, file_path: Path) -> pd.DataFrame:
        logger.info(f"Extracting data from {file_path}")
        return pd.read_csv(file_path)

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        logger.info("Transforming data")
        df = df.drop_duplicates()
        df = df.fillna(0)
        df['processed_at'] = pd.Timestamp.now()
        return df

    def load(self, df: pd.DataFrame, table: str) -> int:
        logger.info(f"Loading {len(df)} rows to {table}")
        df.to_sql(
            table,
            self.engine,
            if_exists='append',
            chunksize=self.batch_size
        )
        return len(df)

    def run(self, file_path: Path, table: str) -> Dict[str, Any]:
        df = self.extract(file_path)
        df = self.transform(df)
        rows = self.load(df, table)
        return {"status": "success", "rows_loaded": rows}

How do I handle large files that don't fit in memory?

10:01 AM

For files larger than memory, use chunked processing with Pandas:

def extract_in_chunks(self, file_path: Path) -> pd.DataFrame:
    logger.info(f"Extracting {file_path} in chunks")
    chunks = pd.read_csv(file_path, chunksize=self.batch_size)
    for chunk in chunks:
        chunk = self.transform(chunk)
        yield chunk

def run_chunked(self, file_path: Path, table: str) -> int:
    total_rows = 0
    for chunk in self.extract_in_chunks(file_path):
        rows = self.load(chunk, table)
        total_rows += rows
    return total_rows

Example with Polars streaming:

import polars as pl

df = pl.scan_csv("large_file.csv")
result = (
    df
    .filter(pl.col("value") > 0)
    .group_by("category")
    .agg(pl.col("value").mean())
    .collect(streaming=True)
)
